1.4 建表 SQL

emp_info 表 


CREATE TABLE `emp_info` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  `md_id` VARCHAR(64) NOT NULL COMMENT '员工主数据ID(业务主键)',
  `user_code` VARCHAR(64) NOT NULL COMMENT '员工工号',
  `name` VARCHAR(128) NOT NULL COMMENT '员工姓名',
  `enable_state_code` INT NULL DEFAULT 2 COMMENT '启用状态编码',
  `enable_state` VARCHAR(32) NULL DEFAULT '已启用' COMMENT '启用状态',
  `deleted` TINYINT(1) NULL DEFAULT 0 COMMENT '是否删除',
  `created_time` DATETIME NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_time` DATETIME NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `data_collection_time` DATETIME NULL COMMENT '数据采集时间',
  PRIMARY KEY (`id`),
  UNIQUE INDEX `uk_md_id` (`md_id`),
  INDEX `idx_user_code` (`user_code`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='外部数据_员工主数据';

emp_leave_record 表 


CREATE TABLE `emp_leave_record` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  `record_id` VARCHAR(64) NOT NULL COMMENT '请休假记录ID',
  `emp_md_id` VARCHAR(64) NULL COMMENT '员工主数据ID',
  `emp_no` VARCHAR(64) NULL COMMENT '员工工号',
  `emp_name` VARCHAR(128) NULL COMMENT '员工姓名',
  `dept_md_id` VARCHAR(64) NULL COMMENT '部门主数据ID',
  `leave_type_name` VARCHAR(128) NULL COMMENT '请休假类型名称',
  `leave_start_date` DATE NULL COMMENT '请休假开始日期',
  `leave_end_date` DATE NULL COMMENT '请休假结束日期',
  `leave_days` DECIMAL(10,2) NULL DEFAULT 0 COMMENT '请休假天数',
  `leave_status_code` INT NULL COMMENT '请休假状态编码',
  `deleted` TINYINT(1) NULL DEFAULT 0 COMMENT '是否删除',
  `data_collection_time` DATETIME NULL COMMENT '数据采集时间',
  PRIMARY KEY (`id`),
  UNIQUE INDEX `uk_record_id` (`record_id`),
  INDEX `idx_emp_md_id` (`emp_md_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='外部数据_员工请休假记录';

emp_org_structure 表 


CREATE TABLE `emp_org_structure` (
  `id` BIGINT NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  `md_id` VARCHAR(64) NOT NULL COMMENT '组织主数据ID',
  `code` VARCHAR(64) NOT NULL COMMENT '组织编码',
  `name` VARCHAR(256) NOT NULL COMMENT '组织名称',
  `simple_name` VARCHAR(128) NULL COMMENT '组织简称',
  `enable_state_code` INT NULL DEFAULT 2 COMMENT '启用状态编码',
  `entity_type` VARCHAR(32) NULL COMMENT '实体组织类型',
  `parent_md_id` VARCHAR(64) NULL COMMENT '上级组织主数据ID',
  `sort_no` INT NULL DEFAULT 0 COMMENT '排序编码',
  `deleted` TINYINT(1) NULL DEFAULT 0 COMMENT '是否删除',
  `data_collection_time` DATETIME NULL COMMENT '数据采集时间',
  PRIMARY KEY (`id`),
  UNIQUE INDEX `uk_md_id` (`md_id`),
  UNIQUE INDEX `uk_code` (`code`),
  INDEX `idx_parent_md_id` (`parent_md_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='外部数据_组织架构主数据';
